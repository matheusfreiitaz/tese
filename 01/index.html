Gerador de Cartão de Perfil do GitHub

Esta ferramenta consome a API pública do GitHub e renderiza dinamicamente os dados do usuário.
Digite qualquer username do GitHub e veja a mágica acontecer! 

// Função JS que busca dados da API do GitHub e atualiza o DOM
async function generateGitHubCard() {
  const username = document.getElementById('githubUsername').value;
  const container = document.getElementById('cardContainer');

  // Exibe um skeleton loader para feedback visual
  container.innerHTML = '<div class="skeleton">Carregando...</div>';
  
  try {
    // Faz a chamada assíncrona para a API
    const response = await fetch(`https://api.github.com/users/${username}`);
    
    if (!response.ok) {
      throw new Error('Usuário não encontrado');
    }
    
    const data = await response.json();
    
    // Manipula o DOM para renderizar o cartão com os dados
    container.innerHTML = `
      <div class="github-card">
        <img src="${data.avatar_url}" ... />
        <div>
          <h3>${data.name || data.login}</h3>
          <p>${data.bio || 'Sem bio.'}</p>
          ...
        </div>
      </div>
    `;

  } catch (error) {
    // Trata o erro e exibe uma mensagem amigável
    container.innerHTML = `<p>Erro: ${error.message}</p>`;
  }
}
